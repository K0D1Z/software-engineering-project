
// Use DBML to define your database structure
// Docs: https://dbml.dbdiagram.io/docs

Table currencies {
  code varchar(3) [primary key]  // ISO 4217 currency symbol: PLN, EUR, USD, GBP etc.
  name varchar(100) [not null]    // Currency name: american dollar, polish złoty etc.
  symbol varchar(5) [not null]   // Currency symbol: zł, $, £ etc.
}

Table users {
  id uuid [primary key]
  client_number varchar(8) [unique, not null]
  pesel varchar(11) [unique, not null]
  first_name varchar(100) [not null]
  last_name varchar(100) [not null]
  email varchar(255) [not null, unique]
  phone_number varchar(20) [not null]
  password_hash varchar(255) [not null]
  created_at timestamp [not null, default: `now()`]
  status varchar(20) [not null, default: 'ACTIVE'] // Enum: ACTIVE, BLOCKED, PENDING, DELETED
}

Table addresses {
  id uuid [primary key]
  user_id uuid [not null, ref: > users.id]
  address_type varchar(20) [not null, default: 'RESIDENTIAL'] // Enum: RESIDENTIAL, CORRESPONDENCE
  street varchar(150) [not null]
  building_number varchar(15) [not null]
  apartment_number varchar(15)
  postal_code varchar(10) [not null]
  city varchar(100) [not null]
  country varchar(2) [not null, default: 'PL'] // ISO 3166-1 alpha-2 (unique 2-letter country id)
  created_at timestamp [not null, default: `now()`]
}

Table accounts {
  id uuid [primary key]
  user_id uuid [ref: > users.id]
  balance numeric(18, 4) [not null, default: 0] // numeric(18, 4) is a standard in banking systems (ISO 4217 & ISO 20022 & Martin Fowler – Patterns of Enterprise Application Architecture)
  currency_code varchar(3) [not null, ref: > currencies.code]
  account_number varchar(34) [unique, not null] // Format: IBAN, up to 34 alphanumeric characters comprising a country code
  account_type varchar(20) [not null] // Enum: CUSTOMER, INTERNAL, SAVING, BUSINESS
  status varchar(20) [not null, default: 'ACTIVE'] // Enum: ACTIVE, FROZEN, REJECTED, CLOSED
  created_at timestamp [not null, default: `now()`]
}

Table transactions {
  id uuid [primary key]
  reference_number varchar(64) [unique, not null] // A transaction reference number (often called a TRN, transaction ID, or UTR) is a unique code assigned to a specific financial transfer or payment.
  description text [not null]
  status varchar(20) [not null, default: 'POSTED'] // Enum: POSTED, PENDING, REVERSED, REJECTED
  executed_at timestamp [not null, default: `now()`]
  reverses_transaction_id uuid [ref: - transactions.id] // Storno auto-reference in case of transaction reversal (immutable ledger)
}

// A double-entry ledger is an accounting record-keeping 
// system where every financial transaction is posted as equal and opposite 
// debit and credit entries across at least two accounts.
Table ledger {
  id uuid [primary key]
  transaction_id uuid [not null, ref: > transactions.id]
  account_id uuid [not null, ref: > accounts.id]
  entry_type varchar(6) [not null] // Enum: DEBIT, CREDIT
  // use numeric type avoid floating point rounding errors and numerical errors
  amount numeric(18, 4) [not null] // numeric(18, 4) is a standard in banking systems (ISO 4217 & ISO 20022 & Martin Fowler – Patterns of Enterprise Application Architecture)
  currency_code varchar(3) [not null, ref: > currencies.code]
  created_at timestamp [not null, default: `now()`]
}

Table exchange_rates {
  id bigserial [primary key] // bigserial is a suitable option because of a large number of entries in this table
  base_currency varchar(3) [not null, ref: > currencies.code]
  target_currency varchar(3) [not null, ref: > currencies.code]
  rate numeric(12, 6) [not null] // numeric(12, 6) is a standard in banking systems (Financial Information eXchange Protocol, NBP, ECB, Forex standard)
  rate_timestamp timestamp [not null, default: `now()`]
}

Table scheduled_transactions {
  id uuid [primary key]
  source_account_id uuid [not null, ref: > accounts.id]
  target_account_id uuid [not null, ref: > accounts.id]
  currency varchar(3) [not null, ref: > currencies.code]
  // use numeric type avoid floating point rounding errors and numerical errors
  amount numeric(18, 4) [not null] // numeric(18, 4) is a standard in banking systems (ISO 4217 & ISO 20022 & Martin Fowler – Patterns of Enterprise Application Architecture)
  cron_expression varchar(50) [not null] // recurring execution of the job; a proposal to use Unix cron syntax for scheduling tasks, e.g., '0 0 1 * *' (monthly)
  next_run_at timestamp [not null]
  status varchar(20) [not null, default: 'ACTIVE'] // Enum: ACTIVE, CANCELLED
  created_at timestamp [not null, default: `now()`]
}
